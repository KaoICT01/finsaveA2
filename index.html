// ==========================================
// FinSave Pro 2026 Enterprise Backend Engine

const SPREADSHEET_ID = "1DxYFSRaKnLvXp4mIyC7mSd75aDF7KWB7aKMNWBxTda8";
const SHEET_NAME = "Savings1";

function doGet(e) {
  try {
    const action = e.parameter.action || "readAll";
    const sheet = getSheet();
    
    if (action === "readAll") {
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) {
        return responseJSON({ success: true, savings: [], goals: [] });
      }
      
      const headers = data[0];
      const rows = data.slice(1);
      
      const savings = [];
      const goals = [];
      
      rows.forEach((row, index) => {
        const type = row[0] || "SAVING";
        if (type === "SAVING") {
          savings.push({
            id: row[1] || String(index + 1),
            amount: Number(row[2]) || 0,
            note: row[3] || "",
            date: row[4] ? formatDate(row[4]) : "",
            time: row[5] || ""
          });
        } else if (type === "GOAL") {
          goals.push({
            id: row[1] || String(index + 1),
            name: row[3] || "Goal",
            targetAmount: Number(row[2]) || 0,
            currentAmount: Number(row[6]) || 0,
            deadline: row[4] ? formatDate(row[4]) : "",
            createdDate: row[5] ? formatDate(row[5]) : ""
          });
        }
      });
      
      return responseJSON({ success: true, savings: savings, goals: goals });
    }
  } catch (error) {
    return responseJSON({ success: false, error: error.toString() });
  }
}

function doPost(e) {
  try {
    const contents = JSON.parse(e.postData.contents);
    const action = contents.action;
    const sheet = getSheet();
    
    if (action === "addSaving") {
      const item = contents.data;
      sheet.appendRow(["SAVING", item.id, item.amount, item.note, item.date, item.time, 0]);
      return responseJSON({ success: true, message: "Saving saved successfully" });
    }
    
    if (action === "deleteSaving") {
      const id = contents.id;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][1]) === String(id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return responseJSON({ success: true, message: "Saving deleted" });
    }
    
    if (action === "addGoal") {
      const item = contents.data;
      sheet.appendRow(["GOAL", item.id, item.targetAmount, item.name, item.deadline, item.createdDate, 0]);
      return responseJSON({ success: true, message: "Goal created successfully" });
    }
    
    if (action === "depositGoal") {
      const goalId = contents.goalId;
      const amount = Number(contents.amount) || 0;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === "GOAL" && String(data[i][1]) === String(goalId)) {
          const current = Number(data[i][6]) || 0;
          sheet.getRange(i + 1, 7).setValue(current + amount);
          break;
        }
      }
      return responseJSON({ success: true, message: "Deposit completed" });
    }

    if (action === "deleteGoal") {
      const id = contents.id;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === "GOAL" && String(data[i][1]) === String(id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return responseJSON({ success: true, message: "Goal deleted" });
    }
    
    return responseJSON({ success: false, error: "Invalid Action" });
  } catch (err) {
    return responseJSON({ success: false, error: err.toString() });
  }
}

function getSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(["TYPE", "ID", "AMOUNT_OR_TARGET", "NOTE_OR_NAME", "DATE_OR_DEADLINE", "TIME_OR_CREATED", "CURRENT_PROGRESS"]);
  }
  return sheet;
}

function formatDate(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  return String(val);
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
